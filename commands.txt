command to record:

rm -rf /home/<username>/.cache/huggingface/lerobot/<dataset_path>            <--- run this if re-recording
python -m lerobot.rl.gym_manipulator --config_path config.json

command to view an episode:

python -m lerobot.scripts.lerobot_dataset_viz   --repo-id <dataset>  --episode-index 0      <--- to view first episode

command to merge repos:

lerobot-edit-dataset \
  --new_repo_id <new_repo> \
  --operation.type merge \
  --operation.repo_ids "['<first_repo>', '<second_repo>']" \
  --push_to_hub true

command to fine-tune the libero checkpoint:

python -m lerobot.scripts.lerobot_train \
  --dataset.repo_id=<dataset_repo> \
  --policy.type=pi05 \
  --policy.pretrained_path=lerobot/pi05_libero_finetuned_v044 \
  --policy.repo_id=<policy_repo> \
  --output_dir=<local_output_path> \
  --policy.device=cuda \
  --batch_size=8 \
  --steps=20000 \
  --save_checkpoint=true \
  --save_freq=5000 \
  --dataset.video_backend=pyav \
  --wandb.enable=true \
  --wandb.project=<project_name>
